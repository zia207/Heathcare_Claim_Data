charlson_comorbidity_index.py 
# Charlson Comorbidity Index (CCI)
# Predicts 10-year mortality from ICD-10

def calculate_cci(diagnoses: list[str]) -> int:
    weights = {
        "MI": (1, ["I21", "I22", "I252"]),
        "CHF": (1, ["I099", "I11", "I50"]),
        "Diabetes": (1, ["E10", "E11", "E13"]),
        "Renal": (2, ["N18", "N19"]),
        "Malignancy": (2, ["C"]),
        "Metastatic": (6, ["C77", "C78", "C79"]),
        "AIDS": (6, ["B20", "B21", "B22"]),
    }
    score = 0
    for condition, (wt, codes) in weights.items():
        if any(d.startswith(tuple(codes))
               for d in diagnoses):
            score += wt
    return score
lace_index.py 
# LACE Index — 30-day readmission risk
# L: length of stay  A: acuity  C: CCI
# E: ER visits in prior 6 months

def calculate_lace(
    los: int,
    acuity: int,   # 0=elective, 1=urgent
    cci: int,
    er_visits: int
) -> int:
    l_score = min(los, 7)       # 0–7
    a_score = acuity * 3          # 0 or 3
    c_score = min(cci, 5)        # 0–5
    e_score = min(er_visits, 4)  # 0–4
    lace = l_score + a_score + c_score + e_score
    risk = ("High" if lace >= 10
            else "Moderate" if lace >= 5
            else "Low")
    return lace, risk

aco_shared_savings.py 
# ACO Shared Savings Calculation

def calc_shared_savings(
    actual_pmpm, benchmark_pmpm, members
):
    actual_total = actual_pmpm * members * 12
    bench_total  = benchmark_pmpm * members * 12
    savings = bench_total - actual_total

    if savings > 0:
        aco_share = savings * 0.60
        result = "Shared Savings"
    else:
        aco_share = savings * 0.40
        result = "Shared Loss"

    return {
        "result": result,
        "gross_savings": savings,
        "aco_share": aco_share
    }
