# Charlson Comorbidity Index (CCI)
def calculate_cci(diagnoses: list[str]) -> int:
    weights = {
        "MI":          (1, ["I21", "I22"]),
        "CHF":         (1, ["I099", "I50"]),
        "Diabetes":    (1, ["E10", "E11"]),
        "Renal":       (2, ["N18", "N19"]),
        "Malignancy":  (2, ["C"]),
        "Metastatic":  (6, ["C77","C78"]),
        "AIDS":        (6, ["B20", "B21"]),
    }
    score = 0
    for _, (wt, codes) in weights.items():
        if any(d.startswith(tuple(codes))
               for d in diagnoses):
            score += wt
    return score
lace_index.py 
# LACE Index — 30-day readmission risk
def calculate_lace(
    los: int, acuity: int,
    cci: int, er_visits: int
) -> tuple:
    l = min(los, 7)
    a = acuity * 3
    c = min(cci, 5)
    e = min(er_visits, 4)
    lace = l + a + c + e
    risk = ("High"     if lace >= 10 else
            "Moderate" if lace >= 5  else
            "Low")
    return lace, risk

aco_shared_savings.py 
# ACO Shared Savings Calculation
def calc_shared_savings(
    actual_pmpm, benchmark_pmpm, members
):
    actual = actual_pmpm * members * 12
    bench  = benchmark_pmpm * members * 12
    savings = bench - actual

    share = 0.60 if savings > 0 else 0.40
    label = ("Shared Savings" if savings > 0
             else "Shared Loss")
    return {
        "result": label,
        "gross": savings,
        "aco_share": savings * share
    }
